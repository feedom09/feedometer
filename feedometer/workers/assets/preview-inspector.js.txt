/*
 * Visual Builder preview inspector.
 *
 * This file intentionally runs as normal browser JavaScript, served by the
 * Worker from /api/builder/preview-inspector.js. Do not turn it into a
 * JavaScript template string: browser syntax checks and browser tests must be
 * able to validate this exact file before deployment.
 */
(() => {
  'use strict';

  if (window.__feedometerPreviewInspector) return;
  window.__feedometerPreviewInspector = true;

  let activeField = 'container';
  let selectedCards = [];

  const css = document.createElement('style');
  css.textContent = [
    '.__fom_selected { outline: 3px solid #10b981 !important; outline-offset: 2px !important; }',
    '.__fom_hover { outline: 2px dashed #0284c7 !important; outline-offset: 2px !important; cursor: crosshair !important; }'
  ].join('\n');
  (document.head || document.documentElement).appendChild(css);

  const cleanText = (value) => String(value || '').replace(/\s+/g, ' ').trim();
  const escapeSelector = (value) => {
    if (window.CSS && typeof window.CSS.escape === 'function') return window.CSS.escape(value);
    return String(value || '').replace(/[^a-zA-Z0-9_-]/g, '\\$&');
  };

  function closestCard(target) {
    if (!target || !target.closest) return null;
    return target.closest('.article-card, article, [class*="card"], [class*="story"], [class*="teaser"], [class*="post"], [class*="item"], li');
  }

  function cardSelector(card) {
    if (card.classList.contains('article-card')) return '.article-card';
    const className = Array.from(card.classList || [])
      .filter((name) => /^[a-zA-Z_-][a-zA-Z0-9_-]*$/.test(name))
      .slice(0, 2)
      .map((name) => '.' + escapeSelector(name))
      .join('');
    return card.tagName.toLowerCase() + className;
  }

  function extractArticle(card) {
    const titleElement = card.querySelector('.card-title, h1, h2, h3, h4, h5, h6, [class*="title"], [class*="headline"]');
    const linkElement = card.matches('a[href]') ? card : card.querySelector('a[href]');
    const imageElement = card.querySelector('img');
    const descriptionElement = card.querySelector('.card-desc, p, [class*="summary"], [class*="excerpt"], [class*="desc"]');
    const dateElement = card.querySelector('.card-date, time, [class*="date"], [class*="time"]');
    const linkValue = card.getAttribute('data-link') || (linkElement && linkElement.href) || window.location.href;

    return {
      title: cleanText(card.getAttribute('data-title') || (titleElement && titleElement.textContent) || card.textContent || 'Untitled Article'),
      link: linkValue,
      image: card.getAttribute('data-image') || (imageElement && (imageElement.currentSrc || imageElement.src)) || '',
      description: cleanText(card.getAttribute('data-desc') || (descriptionElement && descriptionElement.textContent) || ''),
      pubDate: cleanText(card.getAttribute('data-date') || (dateElement && (dateElement.getAttribute('datetime') || dateElement.textContent)) || '')
    };
  }

  function clearSelection() {
    selectedCards.forEach((card) => card.classList.remove('__fom_selected'));
    selectedCards = [];
  }

  function selectCard(card, additive) {
    if (!additive) clearSelection();
    const currentIndex = selectedCards.indexOf(card);
    if (additive && currentIndex >= 0) {
      selectedCards.splice(currentIndex, 1);
      card.classList.remove('__fom_selected');
      return false;
    }
    if (currentIndex < 0) selectedCards.push(card);
    card.classList.add('__fom_selected');
    return true;
  }

  function sendSelection(card, ctrlKey) {
    const item = extractArticle(card);
    window.parent.postMessage({
      // Use an article-specific contract instead of the generic selector event.
      // Other preview tools cannot accidentally replace this concrete item.
      type: 'FEEDOMETER_ARTICLE_CARD_SELECTED_V2',
      selectionProtocol: 'preview-inspector-v2',
      targetField: activeField,
      field: activeField,
      selector: cardSelector(card),
      titleSelector: '.card-title, h1, h2, h3, h4, h5, h6',
      linkSelector: 'a[href]',
      imageSelector: 'img',
      dateSelector: '.card-date, time',
      descriptionSelector: '.card-desc, p, [class*="summary"], [class*="excerpt"]',
      ctrlKey: Boolean(ctrlKey),
      article: item
    }, '*');
  }

  function activateTab(button) {
    const tab = button.getAttribute('data-tab');
    if (tab === null) return;
    document.querySelectorAll('.tab-btn, [data-tab]').forEach((entry) => entry.classList.remove('active'));
    document.querySelectorAll('.tab-panel').forEach((panel) => {
      panel.classList.remove('active');
      panel.style.setProperty('display', 'none', 'important');
    });
    button.classList.add('active');
    const panel = document.querySelector('.tab-panel[data-panel="' + escapeSelector(tab) + '"]');
    if (panel) {
      panel.classList.add('active');
      panel.style.setProperty('display', 'grid', 'important');
    }
  }

  document.addEventListener('click', (event) => {
    const tab = event.target && event.target.closest && event.target.closest('.tab-btn, [data-tab]');
    if (tab) {
      event.preventDefault();
      event.stopImmediatePropagation();
      activateTab(tab);
      return;
    }

    const card = closestCard(event.target);
    if (!card) return;
    event.preventDefault();
    event.stopImmediatePropagation();
    selectCard(card, event.ctrlKey || event.metaKey);
    sendSelection(card, event.ctrlKey || event.metaKey);
  }, true);

  document.addEventListener('mouseover', (event) => {
    const card = closestCard(event.target);
    if (card) card.classList.add('__fom_hover');
  }, true);
  document.addEventListener('mouseout', (event) => {
    const card = closestCard(event.target);
    if (card) card.classList.remove('__fom_hover');
  }, true);

  window.addEventListener('message', (event) => {
    const data = event.data || {};
    if (data.type === 'SET_ACTIVE_FIELD') activeField = data.field || 'container';
    if (data.type === 'FEEDOMETER_DESELECT_ALL') clearSelection();
    if (data.type === 'FEEDOMETER_RESELECT_BY_LINK' && data.link) {
      const selector = '[data-link="' + escapeSelector(data.link) + '"]';
      const card = document.querySelector(selector);
      if (card && selectedCards.indexOf(card) < 0) {
        selectedCards.push(card);
        card.classList.add('__fom_selected');
      }
    }
  });

  window.parent.postMessage({ type: 'FEEDOMETER_INSPECTOR_READY' }, '*');
})();
